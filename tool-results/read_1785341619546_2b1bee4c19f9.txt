    31→// Sub-Rating progress bar helper
    32→function RatingBar({ label, score }: { label: string; score: number }) {
    33→  const pct = (score / 5) * 100;
    34→  return (
    35→    <div className="space-y-1.5">
    36→      <div className="flex justify-between text-xs font-medium">
    37→        <span className="text-white/70">{label}</span>
    38→        <span className="text-gold font-bold font-mono">
    39→          {score.toFixed(1)} / 5.0
    40→        </span>
    41→      </div>
    42→      <div className="w-full h-2 rounded-full bg-white/10 overflow-hidden">
    43→        <motion.div
    44→          initial={{ width: 0 }}
    45→          animate={{ width: `${pct}%` }}
    46→          transition={{ duration: 0.8, ease: 'easeOut' }}
    47→          className="h-full bg-gradient-to-r from-gold to-amber rounded-full"
    48→        />
    49→      </div>
    50→    </div>
    51→  );
    52→}
