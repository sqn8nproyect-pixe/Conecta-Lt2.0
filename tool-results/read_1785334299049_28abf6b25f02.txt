   854→
   855→      {activeTab === 'offers' && (

   910→              <button type="submit" className="px-6 py-2.5 rounded-xl bg-amber-500 hover:bg-amber-400 text-black font-bold text-xs transition-all">

   917→
   918→          {/* Reviews List */}
   919→          <div className="space-y-4">

   927→                  <div className="flex text-amber-400">
   928→                    {[...Array(rev.rating)].map((_, i) => (

   930→                    ))}
   931→                  </div>
   932→                </div>
   933→                <p className="text-xs text-white/70 leading-relaxed">{rev.comment}</p>
   934→              </div>
   935→            ))}
   936→          </div>
   937→        </div>
   938→      )}
   939→    </motion.div>
   940→  );
   941→}
