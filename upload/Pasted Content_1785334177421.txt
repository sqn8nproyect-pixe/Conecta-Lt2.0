// MAIN APP COMPONENT
// ==========================================
export default function App() {
  const [user, setUser] = useState<UserProfile | null>(null);
  const [reviews, setReviews] = useState<Review[]>(mockReviews);
  const [favorites, setFavorites] = useState<number[]>([]);
  const [notifications, setNotifications] = useState<{ id: number; message: string; type: 'success' | 'info' }[]>([]);

  const location = useLocation();

  const addNotification = (message: string, type: 'success' | 'info' = 'success') => {
    const id = Date.now();
    setNotifications(prev => [...prev, { id, message, type }]);
    setTimeout(() => {
      setNotifications(prev => prev.filter(n => n.id !== id));
    }, 4000);
  };

  const toggleFavorite = (id: number) => {
    setFavorites(prev => 
      prev.includes(id) ? prev.filter(fav => fav !== id) : [...prev, id]
    );
    const est = establishments.find(e => e.id === id);
    addNotification(
      favorites.includes(id) ? `Eliminado de favoritos: ${est?.name}` : `¡Añadido a favoritos!: ${est?.name}`
    );
  };

  const loginWithGoogle = () => {
    setUser(mockGoogleUser);
    addNotification("¡Sesión iniciada con éxito vía Google!");
  };

  const logout = () => {
    setUser(null);
    addNotification("Sesión cerrada correctamente.", "info");
  };

  const getDynamicRating = (estId: number) => {
    const estReviews = reviews.filter(r => r.establishmentId === estId);
    const est = establishments.find(e => e.id === estId)!;
    if (!est) return { avg: 4.5, count: 10 };
    if (estReviews.length === 0) return { avg: est.avgRating, count: est.reviewCount };
    
    const totalRating = estReviews.reduce((sum, r) => sum + r.rating, 0);
    const weightedAvg = ((est.avgRating * est.reviewCount) + totalRating) / (est.reviewCount + estReviews.length);
    return { 
      avg: parseFloat(weightedAvg.toFixed(1)), 
      count: est.reviewCount + estReviews.length 
    };
  };

  const addReview = (estId: number, rating: number, comment: string) => {
    if (!user) return false;
    const newReview: Review = {
      id: Date.now(),
      establishmentId: estId,
      userId: user.id,
      userName: user.name,
      userAvatar: user.avatar,
      rating,
      comment,
      date: new Date().toISOString().split('T')[0]
    };
    setReviews([newReview, ...reviews]);
    addNotification("¡Reseña publicada! Gracias por tu opinión.");
    return true;
  };

  return (
    <div className="min-h-screen bg-[#0A0A0C] text-white font-sans relative selection:bg-amber-500 selection:text-black">
      {/* Visual Ambient Orbs */}
      <div className="fixed inset-0 pointer-events-none overflow-hidden z-0">
        <div className="absolute -top-40 -left-40 w-96 h-96 bg-amber-500/10 rounded-full blur-[120px]" />
        <div className="absolute top-1/2 -right-40 w-96 h-96 bg-purple-600/10 rounded-full blur-[140px]" />
      </div>

      {/* Toast Notifications */}
      <div className="fixed top-24 right-6 z-50 flex flex-col gap-3 pointer-events-none">
        <AnimatePresence>
          {notifications.map(n => (
            <motion.div
              key={n.id}
              initial={{ opacity: 0, y: -20, scale: 0.9 }}
              animate={{ opacity: 1, y: 0, scale: 1 }}
              exit={{ opacity: 0, x: 50, scale: 0.9 }}
              className={`pointer-events-auto px-5 py-3.5 rounded-2xl bg-white/10 backdrop-blur-xl border flex items-center gap-3 shadow-2xl ${
                n.type === 'success' ? 'border-emerald-500/40 text-emerald-300' : 'border-amber-500/40 text-amber-300'
              }`}
            >
              <div className={`p-1 rounded-full ${n.type === 'success' ? 'bg-emerald-500/20' : 'bg-amber-500/20'}`}>
                {n.type === 'success' ? <Check size={16} /> : <Sparkles size={16} />}
              </div>
              <span className="text-sm font-medium text-white">{n.message}</span>
            </motion.div>
          ))}
        </AnimatePresence>
      </div>

      {/* Glass Navigation Bar */}
      <nav className="fixed top-0 left-0 right-0 z-40 bg-[#0A0A0C]/80 backdrop-blur-md border-b border-white/10">
        <div className="max-w-7xl mx-auto px-6 h-20 flex items-center justify-between">
          <Link to="/" className="flex items-center gap-3 group">
            <div className="w-10 h-10 rounded-xl bg-amber-500 flex items-center justify-center font-black text-black text-xl shadow-lg shadow-amber-500/20 group-hover:scale-105 transition-transform">
              C
            </div>
            <div>
              <div className="text-xl font-bold tracking-tight text-white font-serif">
                CONECTA<span className="text-amber-400">-LT</span>
              </div>
              <div className="text-[9px] text-white/40 tracking-[2px] font-mono -mt-1">LOS TEQUES</div>
            </div>
          </Link>

          <div className="flex items-center gap-6 text-sm font-medium">
            <Link to="/" className={`transition-colors ${location.pathname === '/' ? 'text-amber-400 font-semibold' : 'text-white/70 hover:text-white'}`}>Directorio</Link>
            <Link to="/mapa" className={`transition-colors ${location.pathname === '/mapa' ? 'text-amber-400 font-semibold' : 'text-white/70 hover:text-white'}`}>Mapa Interactivo</Link>
            
            {user ? (
              <div className="flex items-center gap-3 pl-4 border-l border-white/10">
                <img src={user.avatar} className="w-8 h-8 rounded-full ring-2 ring-amber-500/50" alt="" />
                <span className="hidden md:inline text-sm font-semibold">{user.name.split(" ")[0]}</span>
                <button onClick={logout} className="p-2 text-white/50 hover:text-white rounded-lg hover:bg-white/5 transition-colors">
                  <LogOut size={16} />
                </button>
              </div>
            ) : (
              <button 
                onClick={loginWithGoogle}
                className="flex items-center gap-2 px-4 py-2 rounded-xl bg-amber-500 text-black font-semibold text-xs tracking-wider hover:bg-amber-400 active:scale-95 transition-all shadow-lg shadow-amber-500/10"
              >
                <User size={14} /> ACCEDER CON GOOGLE
              </button>
            )}
          </div>
        </div>
      </nav>

      {/* Main Content View Container */}
      <div className="pt-20 relative z-10">
        <AnimatePresence mode="wait">
          <Routes location={location} key={location.pathname}>
            <Route path="/" element={
              <HomePage 
                establishments={establishments} 
                getDynamicRating={getDynamicRating}
                favorites={favorites}
                toggleFavorite={toggleFavorite}
              />
            } />
            <Route path="/mapa" element={
              <MapPage 
                establishments={establishments} 
                getDynamicRating={getDynamicRating} 
              />
            } />
            <Route path="/establecimiento/:id" element={
              <EstablishmentPage 
                user={user} 
                establishments={establishments} 
                offers={offers} 
                reviews={reviews} 
                addReview={addReview} 
                getDynamicRating={getDynamicRating}
                addNotification={addNotification}
                favorites={favorites}
                toggleFavorite={toggleFavorite}
              />
            } />
          </Routes>
        </AnimatePresence>
      </div>
    </div>
  );
}

                <div className="px-6 pb-6 pt-2">
                  <Link 
                    to={`/establecimiento/${est.id}`}
                    className="w-full py-3 rounded-xl bg-white/10 hover:bg-amber-500 hover:text-black font-semibold text-xs tracking-wider flex items-center justify-center gap-2 transition-all group-hover:shadow-lg"
                  >
                    VER DETALLES & OFERTAS <ChevronRight size={14} />
                  </Link>
                </div>
              </motion.div>
            );
          })}
        </div>
      </div>

      {/* Matchmaker Interactive Modal */}
      <AnimatePresence>
        {matchmakerOpen && (
          <MatchmakerModal

      </AnimatePresence>

  );

// ==========================================
// ESTABLISHMENT DETAIL PAGE
// ==========================================
function EstablishmentPage({ 
  user, establishments, offers, reviews, addReview, getDynamicRating, addNotification, favorites, toggleFavorite 
}: any) {
  const { id } = useParams();
  const navigate = useNavigate();
  const estId = parseInt(id || "1");
  const est = establishments.find((e: Establishment) => e.id === estId) || establishments[0];
  const estOffers = offers.filter((o: Offer) => o.establishmentId === estId);
  const estReviews = reviews.filter((r: Review) => r.establishmentId === estId);
  const { avg, count } = getDynamicRating(estId);

  const [activeTab, setActiveTab] = useState<'info' | 'offers' | 'reviews'>('info');
  const [newRating, setNewRating] = useState(5);
  const [newComment, setNewComment] = useState("");
  const [claimedCode, setClaimedCode] = useState<string | null>(null);

  const isFav = favorites.includes(est.id);

  const handleSubmitReview = (e: React.FormEvent) => {
    e.preventDefault();
    if (!newComment.trim()) return;
    const ok = addReview(estId, newRating, newComment);
    if (ok) setNewComment("");
  };

  return (
    <motion.div 
      initial={{ opacity: 0 }}
      animate={{ opacity: 1 }}
      exit={{ opacity: 0 }}
      className="pb-24 max-w-6xl mx-auto px-6 pt-6"
    >
      <button onClick={() => navigate(-1)} className="flex items-center gap-2 text-xs font-semibold text-white/50 hover:text-white mb-6 transition-colors">
        <ArrowLeft size={16} /> VOLVER AL DIRECTORIO
      </button>

      {/* Header Banner */}
      <div className="relative h-80 rounded-3xl overflow-hidden border border-white/10 mb-8">
        <img src={est.coverImage} alt={est.name} className="w-full h-full object-cover" />
        <div className="absolute inset-0 bg-gradient-to-t from-[#0A0A0C] via-[#0A0A0C]/50 to-transparent" />

        <div className="absolute bottom-6 left-6 right-6 flex flex-col md:flex-row md:items-end justify-between gap-4">
          <div>
            <span className="px-3 py-1 rounded-full bg-amber-500 text-black text-[10px] font-bold uppercase tracking-widest mb-3 inline-block">
              {est.category}
            </span>
            <h1 className="text-3xl md:text-5xl font-serif font-bold text-white">{est.name}</h1>
            <p className="text-sm text-white/70 mt-1 flex items-center gap-2">
              <MapPin size={14} className="text-amber-400" /> {est.address}
            </p>
          </div>

          <div className="flex items-center gap-3">
            <button 
              onClick={() => toggleFavorite(est.id)}
              className={`p-3 rounded-2xl backdrop-blur-md border border-white/20 transition-all ${
                isFav ? 'bg-amber-500 text-black' : 'bg-black/50 text-white'
              }`}
            >
              <Heart size={18} fill={isFav ? "currentColor" : "none"} />
            </button>
            <div className="bg-black/60 backdrop-blur-md px-4 py-2.5 rounded-2xl border border-white/10 flex items-center gap-2 text-amber-400 font-bold text-lg">
              <Star size={18} fill="currentColor" /> {avg}
              <span className="text-xs font-normal text-white/40">({count} opiniones)</span>
            </div>
          </div>
        </div>
      </div>

      {/* Navigation Tabs */}
      <div className="flex gap-4 border-b border-white/10 mb-8">
        {[
          { id: 'info', label: 'Información General' },
          { id: 'offers', label: `Promociones (${estOffers.length})` },
          { id: 'reviews', label: `Reseñas (${estReviews.length})` }
        ].map(tab => (
          <button
            key={tab.id}
            onClick={() => setActiveTab(tab.id as any)}
            className={`pb-4 text-sm font-semibold transition-all relative ${
              activeTab === tab.id ? 'text-amber-400' : 'text-white/50 hover:text-white'
            }`}
          >
            {tab.label}
            {activeTab === tab.id && (
              <motion.div layoutId="tab-underline" className="absolute bottom-0 left-0 right-0 h-0.5 bg-amber-400" />
            )}
          </button>
        ))}
      </div>

      {/* Tab Content */}
      {activeTab === 'info' && (
        <div className="grid grid-cols-1 md:grid-cols-3 gap-8">
          <div className="md:col-span-2 space-y-6">
            <div className="bg-white/5 border border-white/10 p-6 rounded-3xl">
              <h3 className="text-lg font-bold font-serif mb-3">Sobre este local</h3>
              <p className="text-sm text-white/70 leading-relaxed">{est.description}</p>
            </div>

            {/* Sub-Ratings Breakdown */}
            <div className="bg-white/5 border border-white/10 p-6 rounded-3xl space-y-4">
              <h3 className="text-lg font-bold font-serif mb-2">Evaluación del Ambiente</h3>
              <RatingBar label="Ambiente & Decoración" score={est.subRatings.ambiente} />
              <RatingBar label="Calidad del Servicio" score={est.subRatings.servicio} />
              <RatingBar label="Relación Precio / Calidad" score={est.subRatings.precioCalidad} />
            </div>
          </div>

          {/* Contact Details Card */}
          <div className="bg-white/5 border border-white/10 p-6 rounded-3xl space-y-4 h-fit">
            <h3 className="text-lg font-bold font-serif mb-4">Contacto & Horarios</h3>
            <div className="flex items-center gap-3 text-xs text-white/70">
              <Phone size={16} className="text-amber-400" />
              <span>{est.phone}</span>
            </div>
            <div className="flex items-center gap-3 text-xs text-white/70">
              <Instagram size={16} className="text-amber-400" />
              <span>{est.instagram}</span>
            </div>
            <div className="flex items-center gap-3 text-xs text-white/70">
              <Clock size={16} className="text-amber-400" />
              <span>{est.schedule}</span>
            </div>
          </div>
        </div>
      )}

      {activeTab === 'offers' && (
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
          {estOffers.length > 0 ? estOffers.map((off: Offer) => (
            <div key={off.id} className="bg-white/5 border border-white/10 rounded-3xl p-6 flex flex-col justify-between">
              <div>
                <div className="flex items-center justify-between mb-3">
                  <span className="px-3 py-1 rounded-full bg-amber-500/20 border border-amber-500/30 text-amber-400 text-[10px] font-bold">
                    {off.discount}
                  </span>
                  <span className="text-2xl font-bold font-serif text-white">{off.price}</span>
                </div>
                <h4 className="text-lg font-bold mb-1">{off.title}</h4>
                <p className="text-xs text-white/60 mb-4">{off.description}</p>
              </div>

              <button 
                onClick={() => {
                  setClaimedCode(off.code);
                  addNotification(`¡Cupón activado!: ${off.code}`);
                }}
                className="w-full py-3 rounded-xl bg-amber-500 hover:bg-amber-400 text-black font-bold text-xs tracking-wider flex items-center justify-center gap-2 transition-all"
              >
                <Ticket size={16} /> RECLAMAR CÓDIGO {off.code}
              </button>
            </div>
          )) : (
            <p className="text-xs text-white/40 col-span-2 text-center py-12">No hay ofertas activas en este momento.</p>
          )}
        </div>
      )}

      {activeTab === 'reviews' && (
        <div className="space-y-8 max-w-2xl">
          {/* Add Review Form */}
          {user ? (
            <form onSubmit={handleSubmitReview} className="bg-white/5 border border-white/10 p-6 rounded-3xl space-y-4">
              <h4 className="text-sm font-bold tracking-wider uppercase text-amber-400">Escribir una Reseña</h4>
              <div className="flex items-center gap-2">
                {[1, 2, 3, 4, 5].map((star) => (
                  <button
                    key={star}
                    type="button"
                    onClick={() => setNewRating(star)}
                    className="text-amber-400 p-1 hover:scale-110 transition-transform"
                  >
                    <Star size={20} fill={star <= newRating ? "currentColor" : "none"} />
                  </button>
                ))}
              </div>
              <textarea
                value={newComment}
                onChange={(e) => setNewComment(e.target.value)}
                placeholder="¿Qué tal fue tu experiencia en este lugar?"
                className="w-full bg-black/40 border border-white/10 rounded-2xl p-4 text-xs text-white placeholder:text-white/30 outline-none focus:border-amber-400 transition-all h-24 resize-none"
              />
              <button type="submit" className="px-6 py-2.5 rounded-xl bg-amber-500 hover:bg-amber-400 text-black font-bold text-xs transition-all">
                PUBLICAR RESEÑA
              </button>
            </form>
          ) : (
            <p className="text-xs text-white/40 italic">Inicia sesión con Google para dejar tu opinión.</p>
          )}

          {/* Reviews List */}
          <div className="space-y-4">
            {estReviews.map((rev: Review) => (
              <div key={rev.id} className="bg-white/5 border border-white/10 p-5 rounded-2xl space-y-2">
                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-3">
                    <img src={rev.userAvatar} className="w-8 h-8 rounded-full" alt="" />
                    <span className="text-xs font-bold text-white">{rev.userName}</span>
                  </div>
                  <div className="flex text-amber-400">
                    {[...Array(rev.rating)].map((_, i) => (
                      <Star key={i} size={12} fill="currentColor" />
                    ))}
                  </div>
                </div>
                <p className="text-xs text-white/70 leading-relaxed">{rev.comment}</p>
              </div>
            ))}
          </div>
        </div>
      )}
    </motion.div>
  );
}

// Sub-Rating Progress Bar Helper Component
function RatingBar({ label, score }: { label: string; score: number }) {
  return (
    <div className="space-y-1">
      <div className="flex justify-between text-xs font-medium">
        <span className="text-white/70">{label}</span>
        <span className="text-amber-400 font-bold">{score} / 5.0</span>
      </div>
      <div className="w-full h-2 rounded-full bg-white/10 overflow-hidden">
        <div className="h-full bg-amber-500 rounded-full" style={{ width: `${(score / 5) * 100}%` }} />
      </div>
    </div>
  );
}
